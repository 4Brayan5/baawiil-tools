cmake_minimum_required(VERSION 3.24)
project(BaawiilMonitor VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
include(CTest)
option(BAAWIIL_QT_TESTS "Build Qt storage integration tests" OFF)
option(BAAWIIL_UI_TESTS "Build offscreen UI component tests with non-Windows test adapter" OFF)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_include_directories(core_tests PRIVATE src)
  add_test(NAME core COMMAND core_tests)
endif()
if(BUILD_TESTING AND BAAWIIL_UI_TESTS)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Sql Concurrent PrintSupport Network)
  set(CMAKE_AUTOMOC ON)
  add_executable(ui_tests tests/ui_tests.cpp tests/ui_backend_stub.cpp src/window.cpp src/window.h src/collector.h
    src/features/workbench.cpp src/features/workbench.h src/features/history.cpp src/features/updates.cpp src/features/updates.h src/features/settings_io.cpp)
  target_include_directories(ui_tests PRIVATE src)
  target_link_libraries(ui_tests PRIVATE Qt6::Widgets Qt6::Sql Qt6::Concurrent Qt6::PrintSupport Qt6::Network)
  add_test(NAME ui_components COMMAND ui_tests)
  set_tests_properties(ui_components PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(BUILD_TESTING AND BAAWIIL_QT_TESTS)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Sql)
  add_executable(feature_tests tests/feature_tests.cpp src/features/history.cpp src/features/settings_io.cpp)
  target_include_directories(feature_tests PRIVATE src)
  target_link_libraries(feature_tests PRIVATE Qt6::Core Qt6::Sql)
  add_test(NAME features COMMAND feature_tests)
endif()
option(BAAWIIL_BUILD_APP "Build Windows application" ON)
if(BAAWIIL_BUILD_APP)
  if(NOT WIN32)
    message(FATAL_ERROR "Application requires Windows. Use -DBAAWIIL_BUILD_APP=OFF for core tests.")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "A 64-bit toolchain is required.")
  endif()
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent Sql PrintSupport Network)
  qt_standard_project_setup()
  qt_add_executable(baawiil WIN32 src/main.cpp src/window.cpp src/window.h src/collector.cpp src/collector.h
    src/features/backend.cpp src/features/backend.h src/features/history.cpp src/features/history.h
    src/features/workbench.cpp src/features/workbench.h src/features/analysis.h src/features/updates.cpp src/features/updates.h src/features/settings_io.cpp)
  target_compile_definitions(baawiil PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(baawiil PRIVATE Qt6::Widgets Qt6::Concurrent Qt6::Sql Qt6::PrintSupport Qt6::Network psapi pdh iphlpapi ws2_32 advapi32 wintrust crypt32 version)
  qt_add_executable(baawiil-admin WIN32 src/admin_main.cpp src/features/backend.cpp)
  target_compile_definitions(baawiil-admin PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(baawiil-admin PRIVATE Qt6::Widgets psapi pdh iphlpapi ws2_32 advapi32 wintrust crypt32 version shell32)
  target_link_libraries(baawiil PRIVATE shell32 dwmapi)
  qt_add_executable(baawiil-worker src/worker_main.cpp src/features/backend.cpp)
  target_compile_definitions(baawiil-worker PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(baawiil-worker PRIVATE Qt6::Core psapi pdh iphlpapi ws2_32 advapi32 wintrust crypt32 version shell32)
  if(MSVC)
    target_compile_options(baawiil PRIVATE /W4 /utf-8 /permissive-)
    target_compile_options(baawiil-admin PRIVATE /W4 /utf-8 /permissive-)
    target_compile_options(baawiil-worker PRIVATE /W4 /utf-8 /permissive-)
  endif()
  install(TARGETS baawiil baawiil-admin baawiil-worker RUNTIME DESTINATION .)
  qt_generate_deploy_app_script(TARGET baawiil OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
